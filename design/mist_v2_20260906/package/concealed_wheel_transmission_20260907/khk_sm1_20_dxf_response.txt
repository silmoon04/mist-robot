	function requiredValidator(element, trim) {
		var val = "";
		if (element) {
			val += element.value;
		} else {
			return false;
		}
		if (trim) {
			val = Trim(val);
		}
		if (val.length==0) {
			return false;
		}
		return true;
	}

	// Trim関数
	function Trim(value){
		var strRt;
		strRt = LTrim(value);
		strRt = RTrim(strRt);
		return strRt;
	}

	// LTrim関数
	function LTrim(value){
		var i;
		for (i = 0; i < value.length; i ++){
			if (value.charAt(i) != " " && value.charAt(i) != "　"){
				return value.substring( i, value.length);
			}
		}
		return "";
	}

	// RTrim関数
	function RTrim(value){
		var i;
		for (i = value.length - 1; i >= 0; i --){
			if (value.charAt(i) != " " && value.charAt(i) != "　"){
				return value.substring( 0, i + 1);
			}
		}
		return "";
	}


/* クッキー書き込み関数 */
function setCookie(key_name, value){
	var ex = "Mon, 31 Dec 2040 23:59:59 GMT;";
	document.cookie = escape(key_name) + "=" + escape(value) + ";path=/;expires=" + ex;
}

/* クッキー読み込み関数 */
function getCookie(key_name){
	var c_data = "";
	if(key_name){
		c_data = document.cookie + ";";
		c_data = unescape(c_data);
		var n_point = c_data.indexOf(key_name); //（1）

		var v_point = c_data.indexOf("=",n_point) + 1; //（2）

		var end_point = c_data.indexOf(";",n_point); //（3）

		if(n_point > -1){
			c_data = c_data.substring(v_point,end_point);
		}else{
			c_data = "";
		}
	}
	return c_data;
}

/* salt and hash */
function convertPword(user_id, pword){
	// make 'salt' (get SHA256-Hash from 'user_id')
	var shaObj_uid =  new jsSHA(user_id, "ASCII");
	var salt = shaObj_uid.getHash("SHA-256", "HEX");

	// make hashed password
	var shaObj_pass = new jsSHA(salt + pword, "ASCII");
	var sha256digest = shaObj_pass.getHash("SHA-256", "HEX");

	// to do more stretching …

	return sha256digest;
}
